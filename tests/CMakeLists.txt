add_executable(module_smoke smoke.cpp)
target_link_libraries(module_smoke PRIVATE Lidar::ChartCore)
target_link_libraries(module_smoke PRIVATE LidarUICoreplugin LidarChartCoreplugin)
add_executable(laser_telemetry laser-telemetry.cpp)
target_link_libraries(laser_telemetry PRIVATE Lidar::ChartCore LidarUICoreplugin LidarChartCoreplugin)
file(GLOB_RECURSE headers "${PROJECT_SOURCE_DIR}/include/*.h")
set(units)
foreach(header IN LISTS headers)
    file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}/include" "${header}")
    string(MAKE_C_IDENTIFIER "${relative}" unit)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp" "#include <${relative}>\n")
    list(APPEND units "${CMAKE_CURRENT_BINARY_DIR}/${unit}.cpp")
endforeach()
add_library(public_headers OBJECT ${units})
target_link_libraries(public_headers PRIVATE Lidar::ChartCore)
foreach(target module_smoke laser_telemetry public_headers)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endforeach()
add_test(NAME module_smoke COMMAND module_smoke)
add_test(NAME laser_telemetry COMMAND laser_telemetry)
set_tests_properties(module_smoke laser_telemetry PROPERTIES TIMEOUT 30
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=null;QT_FORCE_STDERR_LOGGING=1"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
if(WIN32)
    set_property(TEST module_smoke laser_telemetry APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
else()
    set_property(TEST module_smoke laser_telemetry APPEND PROPERTY ENVIRONMENT "LC_ALL=C.UTF-8")
endif()
